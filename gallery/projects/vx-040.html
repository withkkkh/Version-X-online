寝室值日信息看板

阅读项目文档 ↓

寝室值日信息看板：（1）通过超声传感器检测垃圾桶是否已满，将相关信息通过 ESP-NOW 无线传输至看板端，自动显示并提醒值日生倒垃圾（再也不用担心我不好意思提醒室友倒垃圾了）（2）联网获取时间及天气信息，用于向寝室成员推送相关提醒

项目文档

暑期项目介绍

暑期项目介绍

项目概况

项目名称：寝室日常信息看板

mentor：汪琪峰

项目简介

作为一个日常看板，它能显示一些很日常的信息（

- 和传感器模块远程互联，计算距离显示垃圾桶有没有满，，以及值日生是谁，及时提醒懒惰的室友倒垃圾

- 连接wifi，利用NTP校时显示目前的时间、日期和星期信息

- 连接wifi，调用天气信息API，显示实时天气信息

- 检测有无人靠近，自动亮屏/息屏

后续经过面谈，觉得这个天气信息还不够准。在佳朔哥的建议下去学习了一个通过超声波测距反馈有无雨滴的测下雨装置，信息通过esp-now远程传输给看板，实现（应该是）最精准的下雨检测。但目前时间不够只完成了电路原理图的学习和设计，打板还没打出来

项目成长

硬件选型

MCU：ESP32S（显示屏端）/ESP32C（传感器端）

显示屏：4.3寸陶晶驰串口电容屏

传感器（测距用）：HC-SR04超声传感器

> 最开始想用红外传感器测距，但调研后发现红外传感器测距会受到垃圾袋、玻璃瓶、塑料瓶等折射影响，最后选了超声传感器

>

// WiFi已经连接成功

bool getWeather()

{

WiFiClientSecure client;

client.setInsecure();

HTTPClient http;

String url =

String(host) +

"/weather/v1/current/" +

latitude +

"/" +

longitude +

"?localTime=true&lang=zh";

if (

!http.begin(

client,

url

)

)

{

return false;

}

http.addHeader(

apiKey

);

// 和风天气返回gzip

http.addHeader(

"Accept-Encoding",

"gzip"

);

int httpCode =

http.GET();

if (httpCode != 200)

{

http.end();

return false;

}

// 删除上一次文件

LittleFS.remove(

"/weather.json"

);

// ====================

// Gzip解压

// ====================

GzUnpacker gzip;

bool gzipOK =

gzip.gzStreamExpander(

http.getStreamPtr(),

LittleFS,

"/weather.json"

);

if (!gzipOK)

{

http.end();

return false;

}

http.end();

// ====================

// 打开JSON文件

// ====================

File file =

LittleFS.open(

"/weather.json",

"r"

);

if (!file)

{

return false;

}

// ====================

// JSON解析

// ====================

JsonDocument doc;

DeserializationError error =

deserializeJson(

doc,

file

);

file.close();

if (error)

{

return false;

}

// ====================

// 提取数据

// ====================

int weatherCode =

doc["condition"]["code"]

.as<int>();

int temperature =

round(

doc["temperature"]["value"]

.as<float>()

);

int humidity =

round(

doc["humidity"]

.as<float>() *

100.0f

);

// ====================

// 发给淘晶驰

// ====================

showWeather(

weatherCode

);

delay(20);

showTemperature(

temperature

);

delay(20);

sendASCII(

"rh",

String(humidity) + "%"

);

return true;

}

// ==================== 连接 WiFi ====================

bool connectWiFi()

{

WiFi.mode(WIFI_STA);

WiFi.begin(

ssid,

password

);

unsigned long start =

millis();

while (

WiFi.status() !=

WL_CONNECTED

)

{

if (

millis() - start >

20000

)

{

WiFi.disconnect();

delay(300);

// 恢复ESP-NOW信道

esp_wifi_set_channel(

ESPNOW_CHANNEL,

WIFI_SECOND_CHAN_NONE

);

return false;

}

delay(200);

}

return true;

}

// ==================== 断开 WiFi ====================

void disconnectWiFi()

{

WiFi.disconnect();

delay(300);

// 回到ESP-NOW Channel 1

esp_wifi_set_channel(

ESPNOW_CHANNEL,

WIFI_SECOND_CHAN_NONE

);

}

// ==================== 联网更新 ====================

void updateOnlineData()

{

if (!connectWiFi())

{

return;

}

// WiFi刚连上后，给网络一点稳定时间

delay(2000);

// 1. NTP校时

if (syncTime())

{

updateClock();

}

// 2. 获取天气，失败就重试

bool weatherOK = false;

for (int i = 0; i < 3; i++)

{

if (getWeather())

{

weatherOK = true;

break;

}

// 请求失败后先等2秒再试

delay(2000);

}

// 3. 再多保留1秒，确保TJC命令发完

delay(1000);

// 4. 最后再断WiFi

disconnectWiFi();

}

// ==================== Loop ====================

void loop()

{

unsigned long now =

millis();

// ====================

// ESP-NOW垃圾桶数据

// ====================

if (newData)

{

float distance =

receivedData.distance1;

newData = false;

updateTrash(

distance

);

}

// ====================

// 每30秒刷新屏幕时间

//

// 只读取本地系统时钟

// 不联网

// ====================

if (

now - lastClockUpdate >=

CLOCK_INTERVAL

)

{

lastClockUpdate = now;

updateClock();

}

// ====================

// 每10分钟联网一次

//

// NTP校时

// +

// 获取天气

// ====================

if (

now - lastOnlineUpdate >=

ONLINE_INTERVAL

)

{

lastOnlineUpdate = now;

updateOnlineData();

}

delay(5);

//检测人

static unsigned long lastDistanceCheck = 0;

if (millis() - lastDistanceCheck >= 500)

{

lastDistanceCheck = millis();

float distance = getDistance();

// 1米以内认为有人靠近

if (distance < PERSON_DISTANCE)

{

lastPersonDetected = millis();

if (!screenOn)

{

sendCmd("dim=100");

screenOn = true;

}

}

}

// 90秒没有检测到人

if (

screenOn &&

millis() - lastPersonDetected >= SCREEN_ON_TIME

)

{

sendCmd("dim=0");

screenOn = false;

}

}

{

delayMicroseconds(2);

digitalWrite(trigPin, HIGH);

delayMicroseconds(10);

digitalWrite(trigPin, LOW);

pulseIn(echoPin, HIGH, 30000);

return 999;

return duration * 0.0343f / 2.0f;

}

}

);

{

return false;

}

// ====================

// JSON解析

// ====================

JsonDocument doc;

DeserializationError error =

deserializeJson(

doc,

file

);

file.close();

if (error)

{

return false;

}

// ====================

// 提取数据

// ====================

doc["condition"]["code"]

.as<int>();

);

);

);

}

// ==================== 联网更新 ====================

void updateOnlineData()

{

if (!connectWiFi())

{

return;

}

// WiFi刚连上后，给网络一点稳定时间

delay(2000);

// 1. NTP校时

if (syncTime())

{

updateClock();

}

// 2. 获取天气，失败就重试

bool weatherOK = false;

for (int i = 0; i < 3; i++)

{

if (getWeather())

{

weatherOK = true;

break;

}

// 请求失败后先等2秒再试

delay(2000);

}

// 3. 再多保留1秒，确保TJC命令发完

delay(1000);

// 4. 最后再断WiFi

disconnectWiFi();

}

// ==================== Loop ====================

void loop()

{

unsigned long now =

millis();

// ====================

// ESP-NOW垃圾桶数据

// ====================

if (newData)

{

float distance =

receivedData.distance1;

newData = false;

updateTrash(

distance

);

}

// ====================

// 每30秒刷新屏幕时间

//

// 只读取本地系统时钟

// 不联网

// ====================

if (

now - lastClockUpdate >=

CLOCK_INTERVAL

)

{

lastClockUpdate = now;

updateClock();

}

// ====================

// 每10分钟联网一次

//

// NTP校时

// +

// 获取天气

// ====================

if (

now - lastOnlineUpdate >=

ONLINE_INTERVAL

)

{

lastOnlineUpdate = now;

updateOnlineData();

}

delay(5);

//检测人

static unsigned long lastDistanceCheck = 0;

if (millis() - lastDistanceCheck >= 500)

{

lastDistanceCheck = millis();

float distance = getDistance();

// 1米以内认为有人靠近

if (distance < PERSON_DISTANCE)

{

lastPersonDetected = millis();

if (!screenOn)

{

sendCmd("dim=100");

screenOn = true;

}

}

}

// 90秒没有检测到人

if (

screenOn &&

millis() - lastPersonDetected >= SCREEN_ON_TIME

)

{

sendCmd("dim=0");

screenOn = false;

}

}

电路原理图

1. 看板模块电路图比较简单，只有USB电源模块，以及给串口屏和超声传感器预留的接口

2. 后续又增加了测量下雨模块（暂且这样叫）

主控模块：ESP32-WROOM-32D

超声波收发模块：RCWL-9610 + TCT40-16T/R

使用 40 kHz 超声波换能器进行信号发射和接收，RCWL-9610 用于超声波收发控制，与 ESP32 通过 Trigger/Echo 或 I²C 等信号进行通信，负责测距（测有无雨滴落下）

模拟回波处理模块：OPA2354 双运放

接收到的超声波回波通常幅度较小，尤其是雨点这种不强的反射体，因此通过运算放大器及周围的电阻、电容网络进行交流耦合、放大和滤波，之后经过二极管及 RC 网络进行进一步信号处理

显示与环境检测模块

配置了一块 1.8 英寸 TFT 屏幕，用于显示测距结果、波形或系统状态；同时使用 DHT20 温湿度传感器，通过 I²C 与 ESP32 通信，可获取环境温湿度，并可进一步用于超声波声速补偿，提高测距精度。

人机交互与接口模块

使用旋转编码器及按键实现参数调节、菜单切换等操作，并预留 UART 接口方便串口调试和程序下载。

电源模块

通过 USB Type-C 输入电源，经过 AMS1117-3.3 稳压得到 3.3 V，为 ESP32、传感器和外围数字电路供电，并配置滤波电容保证电源稳定。
